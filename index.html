ChisomDominic::ArchitectureEngine.php 
<?php
// Chisom Dominic — Enterprise PHP Service Provider & API Handler
declare(strict_types=1);

namespace App\Core\Services;

final readonly class OrderProcessingEngine
{
    public function __construct(
        private DatabaseConnectionInterface $db,
        private PaymentGatewayInterface $gateway
    ) {}

    public function executeTransaction(array $payload): TransactionResult
    {
        $sanitizedId = filter_var($payload['id'], FILTER_VALIDATE_INT);
        $amount      = (float) ($payload['total_amount'] ?? 0.00);

        // Atomic Database Transaction with Prepared Statements
        return $this->db->transaction(function() use ($sanitizedId, $amount) {
            $auth = $this->gateway->captureCharge($amount);
            return new TransactionResult(status: 'CONFIRMED', ref: $auth->referenceId);
        });
    }
}
